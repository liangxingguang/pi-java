    "\\$\\{(\\d+|ARGUMENTS|@):-([^}]*)}|\\$\\{@:(\\d+)(?::(\\d+))?}|\\$(ARGUMENTS|@|\\d+)");

public static String substituteArgs(String content, List<String> args) {

        String defaultTarget = m.group(1);

            String value = defaultTarget.equals("@") || defaultTarget.equals("ARGUMENTS")
                ? allArgs
